\begin{frame}{Margin note: a citation or side caveat}
  \begin{withmargin}
    \begin{tdmain}
      Lloyd's algorithm alternates assignment and update until the partition
      stops changing. Each step cannot increase $\mathcal{J}$, so it
      converges, though only to a \textcolor{DHBWred}{local} minimum.
      \par\medskip
      The centroid update is just the per-cluster mean: the choice that
      minimises that cluster's contribution to the objective.
    \end{tdmain}
    \begin{tdmargin}
      Lloyd, 1957; published 1982.
      \par\smallskip
      The margin holds the aside, the citation, the small caveat: present,
      but out of the main line of thought.
    \end{tdmargin}
  \end{withmargin}
\end{frame}

% --- Equation on the 2/3 + 1/3 grid: artefact left, reading right ---
\begin{frame}{Equation: the 2/3 + 1/3 grid}
  \begin{withmargin}
    \begin{tdmain}
      \vspace*{1.2em}
      \[
        \mathcal{J}(C,\mu)
          \;=\; \sum_{k=1}^{K}\;\sum_{x \in C_k}
                \bigl\lVert\, x - \textcolor{DHBWred}{\mu_k}\,\bigr\rVert^{2}
      \]
    \end{tdmain}
    \begin{tdmargin}
      The summed squared distance from every point to \textcolor{DHBWred}{its own
      centroid}. Driving it down is precisely what we mean by a coherent cluster.
    \end{tdmargin}
  \end{withmargin}
\end{frame}

% --- Definition box (titled "Definition: ...") + a reference in the margin ---
\begin{frame}{Definition, with a reference in the margin}
  \begin{withmargin}
    \begin{tdmain}
      \begin{tddef}{Centroid}
        The centroid $\mu_k$ of cluster $C_k$ is the mean of its members:
        the point that minimises the summed squared distance within the cluster.
      \end{tddef}
    \end{tdmain}
    \begin{tdmargin}
      Exactly the per-cluster term of $\mathcal{J}$.
      \par\smallskip
      Lloyd (1982), for the original quantisation argument.
    \end{tdmargin}
  \end{withmargin}
\end{frame}

% --- Introducing a key result: a finding in a block + a small equation, with
%     the citation in the margin (this replaces a plain references list) ---
\begin{frame}{Introducing a key result}
  \begin{withmargin}

      \begin{tdblock}
        Density-based clustering recovers groups of \textcolor{DHBWred}{arbitrary}
        shape and marks sparse points as noise, without fixing the number of
        clusters in advance.

    \begin{tdmargin}

      A density-based algorithm for discovering clusters. \textit{Proc.\ KDD}.
    \end{tdmargin}
  \end{withmargin}
\end{frame}

\begin{frame}{Two columns: a side-by-side comparison}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \textbf{Lloyd's algorithm}
      \begin{itemize}
        \item Assign each $x$ to nearest $\mu_k$.
        \item Recompute each $\mu_k$.
        \item Repeat until assignments settle.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Caveats}
      \begin{itemize}
        \item Converges to a \textcolor{DHBWred}{local} minimum.
        \item Sensitive to initialisation.
        \item Assumes roughly spherical clusters.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Pseudocode, with a margin note}
  \begin{withmargin}
    \begin{tdmain}
      \begin{algorithm}[H]
        \DontPrintSemicolon
        \KwIn{data $X$, cluster count $K$}
        \KwOut{assignments $C$, centroids $\mu$}
        initialise $\mu_1,\dots,\mu_K$\;
        \Repeat{assignments stop changing}{
          assign each $x$ to its nearest centroid $\mu_k$\;
          recompute each $\mu_k$ as the mean of its members\;
        }
        \caption{Lloyd's algorithm for $k$-means}
      \end{algorithm}
    \end{tdmain}
    \begin{tdmargin}
      The two loop lines are the whole idea: \textcolor{DHBWred}{assign}, then
      update. Everything else is bookkeeping.
    \end{tdmargin}
  \end{withmargin}
\end{frame}

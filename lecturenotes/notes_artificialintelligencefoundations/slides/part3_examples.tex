%% Part III -- Examples (comic pages 20 to 27).
%% Seven systems that are already running, ordered from the one nobody
%% notices (Maps) to the one that is strapped to a body (exoskeletons). The
%% section is deliberately mundane: the argument is that the deployment
%% already happened.
%% Fragment only: no preamble, \input by ai_for_all.tex.

\sectionsubtitle{pages 20 to 27}
\section{Examples}

% Page 21. The first example in the book is the one the room used this
% morning, and the passage names the input, the inference and the latency.
\begin{frame}{Maps}
  \panel{page 21, two phones and the delta}
  \begin{comicquote}{21}
    Using location data from smartphones\ldots{} the AI-application Maps
    can analyze how fast traffic is moving\ldots{} in real-time.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 22 goes from recommending a product to generating one. The second half
% is the part worth arguing about, so that is the half quoted.
\begin{frame}{Recommendation}
  \panel{page 22, the component tag cloud}
  \begin{comicquote}{22}
    Also, the platforms use the knowledge of our habits to create and offer
    new products similar to previous successes\ldots{} or to let AI
    autonomously stir those components together that have worked great
    elsewhere.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 23 ends on WeChat and state supervision of groups of twelve or more.
% The quote taken is the opening one, because the invisibility is the
% precondition for everything the rest of the page describes.
\begin{frame}{Social Networks}
  \panel{page 23, what a wonderful match}
  \begin{comicquote}{23}
    Those of us who use social networks use AI frequently, without
    necessarily knowing it.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 24 refuses the word autonomous for what ships today, then names the
% real blocker, which is not control but explanation.
\begin{frame}{Self-driving Cars}
  \panel{page 24, the unforeseen circumstance}
  \begin{comicquote}{24}
    Moreover, user tests show that we need autonomous systems to explain
    their decision to feel safe. Not easy.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 25 runs from the 2015 ImageNet crossover to cancer detection to the
% military, and then to adversarial examples. The Picasso panel next to the
% rifle is the one to show.
\begin{frame}{Identifying Images}
  \panel{page 25, rifle 2017 and Picasso 1910}
  \begin{comicquote}{25}
    Unfortunately, AI is susceptible to hallucinations: caused by biased
    data or by hackers who, for example, make AI mistake guns for apes.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 26. The number is what makes the case: if words carry a tenth of the
% meaning, a text-only interface is reading a tenth of the message.
\begin{frame}{Emotional AI}
  \panel{page 26, the tone of voice panel}
  \begin{comicquote}{26}
    Only $\sim$ 10\% of the emotional meaning of a message is conveyed
    through words.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 27, and the one example in the section where the argument is not about
% scale but about fit: a body is a moving target, so the controller has to
% learn rather than be tuned once.
\begin{frame}{Exoskeletons}
  \panel{page 27, elderly man or firefighter}
  \begin{comicquote}{27}
    Our bodies are different and constantly changing. Nowadays, the only
    feasible way of translating this to robotics is through the help of AI.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

%% Part IV -- Chances (comic pages 28 to 36).
%% Eight arguments for the technology, and the book makes them properly: the
%% scale argument, the accuracy argument, the bias argument turned around,
%% and then the quiet one on page 36 that a chance also costs something.
%% Fragment only: no preamble, \input by ai_for_all.tex.

\sectionsubtitle{pages 28 to 36}
\section{Chances}

% Page 29. The bottleneck is not storage, it is us. Everything after this
% page in the section follows from that sentence.
\begin{frame}{Dealing with Big Data}
  \panel{page 29, hundreds of analysts}
  \begin{comicquote}{29}
    Even with hundreds of us analyzing patterns in big data, the sheer
    volume simply overwhelms our capacities.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 30, the lawyer experiment: 20 experienced lawyers, 5 contracts, 30
% issues, 4 hours. The four numbers are the frame; quoting the summary keeps
% all four on the slide.
\begin{frame}{Efficiency}
  \panel{page 30, twenty lawyers against one AI}
  \begin{comicquote}{30}
    Humans achieved, on average, an 85\% accuracy rate, the AI 95\%. Humans
    took 92 minutes to complete the task, the AI\ldots{} 26 seconds.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 31 is the sharpest reasoning in the book. A bias is not an error bar:
% shared distortions do not average out, they compound. That is the argument
% for a second opinion that is not another human.
\begin{frame}{Cognitive Biases}
  \panel{page 31, heuristics and gut feeling}
  \begin{comicquote}{31}
    And this is what makes our distortions have large-scale consequences.
    Were they random, they would cancel out.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 32 gives the machine the generation and keeps the judgement. The last
% clause is the whole position, and it has not aged.
\begin{frame}{Creativity}
  \panel{page 32, just a generic style transfer}
  \begin{comicquote}{32}
    We can also get creative directly with the AI, compiling data and coding
    it, in a narrow area. But the AI has no judgement.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 33, doctors. The chance here is not diagnosis, it is the time the
% diagnosis frees up, and the passage is unusually blunt about what that time
% is currently spent on.
\begin{frame}{Connection}
  \panel{page 33, the overpaid bookkeeper}
  \begin{comicquote}{33}
    Today, they often act like overpaid bookkeepers: taking in and spitting
    back data, prescribing drugs, adjusting doses, ordering tests.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 34, robotics without reprogramming. Demonstration replaces the rule
% set, which is the same move as page 15 applied to a body.
\begin{frame}{New Opportunities}
  \panel{page 34, the sensor jacket and glove}
  \begin{comicquote}{34}
    AI now enables us to teach robots how to move, either together with the
    robot, or, with jackets and gloves containing sensors.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 35. Accessibility is the least contested case in the book, and the
% page lists reading, hearing, typing, the home, the classroom and the ward.
\begin{frame}{Inclusion}
  \panel{page 35, a wall missing several bricks}
  \begin{comicquote}{35}
    AI reads to us, it narrates the world around us and help us `see'.
    Speech-to-text AI helps us `hear', too.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 36 closes the section by undercutting it. The chance is convenience;
% the bill is the analogue institution where people met by accident. Note
% that this is filed under Chances, not Risks, on purpose.
\begin{frame}{Comfort}
  \panel{page 36, mall\_blueprint.pdf}
  \begin{comicquote}{36}
    It is precisely these institutions that are social places where we can
    come together and exchange ideas. Where we can decelerate. `Waste' time.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

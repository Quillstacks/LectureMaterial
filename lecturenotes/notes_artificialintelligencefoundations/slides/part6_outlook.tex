%% The section that stops describing and starts demanding: a device that
%% augments rather than entertains, women in the room, AI read as an economic

\sectionsubtitle{pages 44 to 49}
\section{Outlook}

% Page 45. The budget follows the purchase decision, the purchase decision
% follows entertainment, so the device we got is the device we bought. Read
% the passage as a product-management claim, because that is what it is.
\begin{frame}{Overall Human-Machine Intelligence}
  \panel{page 45, the smartphone as marketplace}
  \begin{comicquote}{45}
    Up to date, there is no smartphone that is especially good at helping us
    get better ideas or interpret information. There is no interface for
    that either.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 46. The page opens on the human computers who ran the first
% calculations, and closes on what a homogeneous development team reproduces.
% The parenthesis in the quote is the authors' own.
\begin{frame}{Feminism}
  \panel{page 46, the first computers were women}
  \begin{comicquote}{46}
    We need (different!) women to participate in the development and use of
    AI. Otherwise we'll just reproduce patriarchal structures.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 47 reads AI through Weber's judge as automaton and Marx's worker as
% appendage of the machine, then names the ask. Quoted here is the ask, not
% the theory, because the theory needs the page's own panels to land.
\begin{frame}{Digital Capitalism}
  \panel{page 47, Weber's judge and Marx's worker}
  \begin{comicquote}{47}
    Since today's AI is largely incomprehensible to us, we need to regulate
    and shape AI as a public service and snatch the omnipotence over AI from
    the few dominant, digital companies.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 48, after Virginia Dignum, and the one page in the book that carries
% two frames. First the definition, because the acronym is worth having
% precisely.
\begin{frame}{ART: Accountability, Responsibility, Transparency}
  \panel{page 48, the three equal pillars}
  \begin{comicquote}{48}
    Accountability is the concept that AI should be held responsible for the
    results of its algorithms.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Second frame on page 48: the demand the definition makes. Splitting the
% page here is deliberate, because a room will nod at the acronym and skip
% the sentence that says the acronym is not enough.
\begin{frame}{ART: Accountability, Responsibility, Transparency}
  \panel{page 48, training, regulation, awareness}
  \begin{comicquote}{48}
    But let's not fool ourselves. For this ART of AI, we need a new and more
    ambitious form of governance.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 49 closes the Outlook on numeracy: none of the demands above can be
% argued by people whose brains signal pain at a number. Zero as the worked
% example is the book's own choice.
\begin{frame}{No Fear of Numbers}
  \panel{page 49, 1908 against MCMVIII}
  \begin{comicquote}{49}
    With improved `math-esteem', more of us could and should discuss chances
    and risks of AI in our lives.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

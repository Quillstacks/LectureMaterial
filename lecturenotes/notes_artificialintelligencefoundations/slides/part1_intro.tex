%% Part I -- Intro (comic pages 6 to 10).
%% The section that establishes why the book exists: the fear we inherited
%% from fiction, the two authors and what each of them is after, and the
%% turn from replication to augmentation that the rest of the book runs on.
%% Fragment only: no preamble, \input by ai_for_all.tex.

\sectionsubtitle{pages 6 to 10}
\section{Intro}

% The book's opening line, and the only frame in the deck whose passage is
% not tied to a topic page. It sets the whole posture: forecasting is not the
% job, choosing is.
\begin{frame}{Knowing what future we would like to have}
  \panel{page 6, the opening spread}
  \begin{comicquote}{6}
    Knowing the future is impossible. But what we can do: Knowing what
    future we would like to have. And then work on it.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 7 runs Golem and Frankenstein into the take-over scenario and lands on
% Hawking. The Hawking pair is quoted whole because the second sentence is
% what makes the first one a problem rather than a boast.
\begin{frame}{Our creatures have always turned against us}
  \panel{page 7, Golem and Frankenstein}
  \begin{comicquote}{7}
    In 2011, Stephen Hawking said that `Success in creating AI would be the
    biggest event in history.' `Unfortunately, it might also be the last,
    unless we learn how to avoid the risks.'
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 8, Julia, the economist. The line lands the bias argument the whole
% Risks section later depends on, and it lands it as a compliment to data.
\begin{frame}{Who is Julia}
  \panel{page 8, Julia and the digit block}
  \begin{comicquote}{8}
    I love data's friendliness, their lack of strategic answering. If they
    are biased it's not their fault but ours.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 9, Lena, the illustrator. Why a comic and not a paper: the subject is
% a mirror, so the medium has to be one too.
\begin{frame}{Who is Lena}
  \panel{page 9, Lena and the visual codes}
  \begin{comicquote}{9}
    AI is a field that gives us an intimate view on society's desires and
    fears.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

% Page 10 parades Maschinen-Maria, R2D2, T-800, Samantha and Tars, then
% closes on the sentence that turns the whole parade into an argument.
\begin{frame}{Lessons from AI in Fiction}
  \panel{page 10, the five fictional robots}
  \begin{comicquote}{10}
    Seems, like we should not seek to replicate but to augment ourselves.
  \end{comicquote}
  \slidesources{Schneider2019}
\end{frame}

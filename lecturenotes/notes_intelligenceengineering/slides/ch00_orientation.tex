% The brainstorm in the big group, in the layout of the Extra Mile frame:
% strip, one block, one closing line. The steps run in order, so they are
% enumerated; 90 minutes in all (10 + 20 + 5 + 10 + 5 + 10 + 15 + 15), 5
% minutes per idea in every writing round. 4 + 2 + 2 =
% eight ideas per player; a stack moves twice, so three tables write on it.
\begin{frame}{Brainstorm together}
  \sideimagecover[0.33]{bees}
  \sidecaption{The bee}
  \begin{sidebody}
    \begin{block}{Brainwriting in the big group, 90 minutes}
      \begin{enumerate}\setlength{\itemsep}{0pt}
        \item \textbf{10 min} Research: look it up online or ask a colleague.
        \item \textbf{20 min} Each player writes four ideas.
        \item \textbf{5 min} Research.
        \item \textbf{10 min} Stacks pass to the next table, each adds two.
        \item \textbf{5 min} Research.
        \item \textbf{10 min} Stacks pass again, each adds the last two.
        \item \textbf{15 min} Cluster and clean the ideas per group.
        \item \textbf{15 min} Cluster across teams.
      \end{enumerate}
    \end{block}
    \alertbf{Eight ideas} per player, every stack read at three tables.
  \end{sidebody}
\end{frame}
